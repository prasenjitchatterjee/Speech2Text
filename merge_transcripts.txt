    "    \"\"\"List of (lower_word, start_char, end_char) for each word in text.\"\"\"\n",
    "    return [(m.group(0).lower(), m.start(), m.end())\n",

    "def _build_ngrams(words: List[str], min_len: int, max_len: int):\n",
    "    \"\"\"dict: n -> set of n-gram tuples (lowercased).\"\"\"\n",
    "    ngrams = {}\n",
    "    max_len = max(min(max_len, len(words)), min_len)\n",
    "    for n in range(min_len, max_len + 1):\n",
    "        s = set()\n",
    "        for i in range(0, len(words) - n + 1):\n",
    "            s.add(tuple(words[i:i+n]))\n",
    "        if s:\n",
    "            ngrams[n] = s\n",
    "    return ngrams\n",
    "\n",
    "def _merge_intervals(intervals: List[Tuple[int,int]]) -> List[Tuple[int,int]]:\n",
    "    if not intervals: return intervals\n",
    "    intervals.sort()\n",
    "    merged = [intervals[0]]\n",
    "    for s,e in intervals[1:]:\n",
    "        ls, le = merged[-1]\n",
    "        merged[-1] = (ls, max(le, e)) if s <= le else merged.append((s,e)) or merged[-1]\n",

    "from typing import List, Dict, Any, Optional\n",
    "\n",
    "def expand_missing_segments(\n",
    "    full_text: str,\n",
    "    segments: List[Dict[str, Any]],\n",
    "    min_words: int = 2,  # default 2 to avoid trivial single-word matches\n",
    ") -> List[Dict[str, Any]]:\n",
    "    \"\"\"\n",
    "    Replace any segment with start==-1 by a series of fragments found sequentially\n",
    "    in full_text using str.find(), matching ONLY on WHOLE-WORD sequences.\n",
    "\n",
    "    - Search window per missing segment: [prev_end, next_known_start)\n",
    "    - Greedy: for remaining words, find the LONGEST subsequence starting at the current\n",
    "      word that appears in the window; advance past it; continue with the remainder.\n",
    "    - If no ≥min_words subsequence matches at a position, skip one word and retry.\n",
    "    - If nothing matches for that segment, keep it as unresolved (start/end stay -1).\n",
    "    - Output is flattened (no nested lists).\n",
    "    \"\"\"\n",
    "\n",
    "    def next_known_start_from_input(idx: int) -> Optional[int]:\n",
    "        for k in range(idx + 1, len(segments)):\n",
    "            st = segments[k].get(\"start\", -1)\n",
    "            if isinstance(st, int) and st != -1:\n",
    "                return st\n",
    "        return None\n",
    "\n",
    "    def last_known_end_from_output(out: List[Dict[str, Any]]) -> int:\n",
    "        for k in range(len(out) - 1, -1, -1):\n",
    "            item = out[k]\n",
    "            if isinstance(item, dict) and isinstance(item.get('end', None), int) and item['end'] != -1:\n",
    "                return item['end']\n",
    "        return 0\n",
    "\n",
    "    def match_fragments_in_window(text: str, window_start: int, window_end: Optional[int]) -> List[Dict[str, Any]]:\n",
    "        words = text.split()\n",
    "        i = 0\n",
    "        frags: List[Dict[str, Any]] = []\n",
    "        search_pos = window_start\n",
    "\n",
    "        while i < len(words):\n",
    "            found_any = False\n",
    "            # try longest -> shortest subsequence of words[i:]\n",
    "            for j in range(len(words), i + min_words - 1, -1):\n",
    "                candidate = \" \".join(words[i:j])\n",
    "                idx = full_text.find(candidate, search_pos) if window_end is None else full_text.find(candidate, search_pos, window_end)\n",
    "                if idx != -1:\n",
    "                    start_idx = idx\n",
    "                    end_idx = idx + len(candidate)\n",
    "                    frags.append({\"text\": candidate, \"start\": start_idx, \"end\": end_idx})\n",
    "                    search_pos = end_idx\n",
    "                    i = j\n",
    "                    found_any = True\n",
    "                    break\n",
    "            if not found_any:\n",
    "                # Skip this word and try again from the next word\n",
    "                i += 1\n",
    "\n",
    "        return frags\n",
    "\n",
    "    output: List[Dict[str, Any]] = []\n",
    "    for i, seg in enumerate(segments):\n",
    "        st = seg.get('start', -1)\n",
    "        if isinstance(st, int) and st != -1:\n",
    "            output.append(seg)\n",
    "            continue\n",
    "\n",
    "        prev_end = last_known_end_from_output(output)\n",
    "        next_start = next_known_start_from_input(i)\n",
    "\n",
    "        frags = match_fragments_in_window(seg.get('text', ''), prev_end, next_start)\n",
    "        if frags:\n",
    "            output.extend(frags)  # flattened\n",
    "        else:\n",
    "            # keep unresolved, as it truly doesn't exist in the window\n",
    "            output.append(seg)\n",
    "\n",
    "    return output\n"
