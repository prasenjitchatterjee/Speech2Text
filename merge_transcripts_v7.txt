    "import re\n",
    "import unicodedata\n",
    "\n",
    "\n",
    "# ---------- helpers ----------\n",
    "\n",
    "def strip_diacritics(s: str) -> str:\n",
    "    return ''.join(\n",
    "        c for c in unicodedata.normalize('NFD', s)\n",
    "        if unicodedata.category(c) != 'Mn'\n",
    "    )\n",
    "\n",
    "\n",
    "def normalize_token(token: str) -> str:\n",
    "    \"\"\"\n",
    "    Lowercase, remove diacritics, strip punctuation.\n",
    "    Used only for matching, never for output text.\n",
    "    \"\"\"\n",
    "    s = strip_diacritics(token.lower())\n",
    "    # keep letters/digits/underscore, drop punctuation\n",
    "    s = re.sub(r'[^\\w]+', '', s)\n",
    "    return s\n",
    "\n",
    "\n",
    "def contains_cjk(text: str) -> bool:\n",
    "    for ch in text:\n",
    "        if '\\u4e00' <= ch <= '\\u9fff' or '\\u3400' <= ch <= '\\u4dbf':\n",
    "            return True\n",
    "    return False\n",
    "\n",
    "\n",
    "def tokenize_with_spans(text: str, mode: str = \"auto\"):\n",
    "    \"\"\"\n",
    "    Returns list of tokens with:\n",
    "        {\n",
    "          'text': original_token_text,\n",
    "          'start': start_char_index_in_text,\n",
    "          'end': end_char_index_in_text,\n",
    "          'norm': normalized_token_for_matching\n",
    "        }\n",
    "    - For non-Chinese → word-level (split on whitespace)\n",
    "    - For Chinese/CJK  → char-level (non-space chars)\n",
    "    \"\"\"\n",
    "    if mode == \"auto\":\n",
    "        mode = \"char\" if contains_cjk(text) else \"word\"\n",
    "\n",
    "    tokens = []\n",
    "\n",
    "    if mode == \"word\":\n",
    "        for m in re.finditer(r'\\S+', text):\n",
    "            tok = m.group()\n",
    "            tokens.append({\n",
    "                \"text\": tok,\n",
    "                \"start\": m.start(),\n",
    "                \"end\": m.end() - 1,\n",
    "                \"norm\": normalize_token(tok),\n",
    "            })\n",
    "    else:  # char mode (e.g., Chinese)\n",
    "        for i, ch in enumerate(text):\n",
    "            if ch.isspace():\n",
    "                continue\n",
    "            tokens.append({\n",
    "                \"text\": ch,\n",
    "                \"start\": i,\n",
    "                \"end\": i,\n",
    "                \"norm\": normalize_token(ch),\n",
    "            })\n",
    "\n",
    "    return tokens\n",
    "\n",
    "\n",
    "def find_best_run(left_tokens, full_tokens, i_left, full_cursor):\n",
    "    \"\"\"\n",
    "    For left_tokens starting at index i_left, and searching full_tokens\n",
    "    from index full_cursor onward, find the BEST run:\n",
    "\n",
    "    - Matching is done on .norm (case/punct/diacritics agnostic).\n",
    "    - Run is a sequence of tokens where:\n",
    "        left[i_left + k].norm == full[j + k].norm\n",
    "      for k = 0..run_len-1\n",
    "    - Returns (run_len, full_start_index) or None if no match.\n",
    "    \"\"\"\n",
    "    L = len(left_tokens)\n",
    "    F = len(full_tokens)\n",
    "    best = None  # (run_len, full_start_index)\n",
    "\n",
    "    tok_norm = left_tokens[i_left][\"norm\"]\n",
    "    if not tok_norm:\n",
    "        return None  # nothing to match\n",
    "\n",
    "    for j in range(full_cursor, F):\n",
    "        if full_tokens[j][\"norm\"] != tok_norm:\n",
    "            continue\n",
    "\n",
    "        # Try to extend contiguous run\n",
    "        l = 0\n",
    "        while i_left + l < L and j + l < F:\n",
    "            ln = left_tokens[i_left + l][\"norm\"]\n",
    "            fn = full_tokens[j + l][\"norm\"]\n",
    "            if not ln or not fn or ln != fn:\n",
    "                break\n",
    "            l += 1\n",
    "\n",
    "        if l > 0:\n",
    "            if best is None or l > best[0] or (l == best[0] and j < best[1]):\n",
    "                best = (l, j)\n",
    "\n",
    "    return best\n",
    "\n",
    "\n",
    "def align_one_entry(full_tokens, entry):\n",
    "    \"\"\"\n",
    "    Align a single left/right entry (one dict with 'text', 'start_time', 'end_time')\n",
    "    against the full_tokens.\n",
    "\n",
    "    Returns list of dicts with:\n",
    "        start_time, end_time, start_pos, end_pos, text\n",
    "    where text is always a substring of entry[\"text\"].\n",
    "    \"\"\"\n",
    "    entry_text = entry[\"text\"]\n",
    "    left_tokens = tokenize_with_spans(entry_text)\n",
    "    L = len(left_tokens)\n",
    "\n",
    "    segments = []  # each: { left_start, left_end, full_start, full_end or None }\n",
    "    i = 0\n",
    "    full_cursor = 0\n",
    "    unmatched_start = None\n",
    "\n",
    "    while i < L:\n",
    "        best = find_best_run(left_tokens, full_tokens, i, full_cursor)\n",
    "\n",
    "        if best is None:\n",
    "            # This token can't be matched → part of an unmatched block\n",
    "            if unmatched_start is None:\n",
    "                unmatched_start = i\n",
    "            i += 1\n",
    "            continue\n",
    "\n",
    "        run_len, full_start = best\n",
    "\n",
    "        # Flush any unmatched block before this match\n",
    "        if unmatched_start is not None and unmatched_start < i:\n",
    "            segments.append({\n",
    "                \"left_start\": unmatched_start,\n",
    "                \"left_end\": i - 1,\n",
    "                \"full_start\": None,\n",
    "                \"full_end\": None,\n",
    "            })\n",
    "        unmatched_start = None\n",
    "\n",
    "        # Add matched block: contiguous in full_tokens\n",
    "        segments.append({\n",
    "            \"left_start\": i,\n",
    "            \"left_end\": i + run_len - 1,\n",
    "            \"full_start\": full_start,\n",
    "            \"full_end\": full_start + run_len - 1,\n",
    "        })\n",
    "\n",
    "        i += run_len\n",
    "        full_cursor = full_start + run_len  # move ahead in full_text as required\n",
    "\n",
    "    # Flush trailing unmatched block\n",
    "    if unmatched_start is not None and unmatched_start < L:\n",
    "        segments.append({\n",
    "            \"left_start\": unmatched_start,\n",
    "            \"left_end\": L - 1,\n",
    "            \"full_start\": None,\n",
    "            \"full_end\": None,\n",
    "        })\n",
    "\n",
    "    # Convert segments to final output dicts\n",
    "    out = []\n",
    "    for seg in segments:\n",
    "        ls, le = seg[\"left_start\"], seg[\"left_end\"]\n",
    "        seg_text = entry_text[left_tokens[ls][\"start\"]: left_tokens[le][\"end\"] + 1]\n",
    "\n",
    "        if seg[\"full_start\"] is None:\n",
    "            start_pos = -1\n",
    "            end_pos = -1\n",
    "        else:\n",
    "            fs, fe = seg[\"full_start\"], seg[\"full_end\"]\n",
    "            start_pos = full_tokens[fs][\"start\"]\n",
    "            end_pos = full_tokens[fe][\"end\"]\n",
    "\n",
    "        out.append({\n",
    "            \"start_time\": entry[\"start_time\"],\n",
    "            \"end_time\": entry[\"end_time\"],\n",
    "            \"start_pos\": start_pos,\n",
    "            \"end_pos\": end_pos,\n",
    "            \"text\": seg_text,\n",
    "        })\n",
    "\n",
    "    return out\n",
    "\n",

    "# ---------- main function ----------\n",
    "\n",
    "def align_channel(full_channel, side_channel, parent_preference: str = \"right\"):\n",
    "    \"\"\"\n",
    "    full_channel : list[{'start_time', 'end_time', 'text', 'channel'=='full'}]\n",
    "        Combined into full_text with a space between segments.\n",
    "\n",
    "    side_channel : list[{'start_time', 'end_time', 'text', 'channel' in {'left','right'}}]\n",
    "\n",
    "    parent_preference:\n",
    "        \"right\" (default) → prefer nearest matched segment to the RIGHT, then LEFT\n",
    "        \"left\"            → prefer nearest matched segment to the LEFT, then RIGHT\n",
    "\n",
    "    Returns:\n",
    "        list of dicts:\n",
    "            {\n",
    "              'channel': 'left' or 'right',\n",
    "              'start_time': ...,\n",
    "              'end_time': ...,\n",
    "              'start_pos': <char index in full_text or -1>,\n",
    "              'end_pos': <char index in full_text or -1>,\n",
    "              'text': <substring of side_channel's text>,\n",
    "              'seg_id': <e.g. 'left_3' or 'left_3_2'>,\n",
    "              'seg_parent_id': -1 or another seg_id string\n",
    "            }\n",
    "\n",
    "    - Matching ignores case, punctuation, diacritics.\n",
    "    - Positions (start_pos/end_pos) are in the original full_text.\n",
    "    - Never injects text from full_channel into output.\n",
    "    - Splits side_channel entries when needed.\n",
    "    - seg_id/seg_parent_id are assigned per the rules described.\n",
    "    \"\"\"\n",
    "    # IMPORTANT: space-join to avoid gluing last & first words\n",
    "    full_text = \" \".join(fc[\"text\"] for fc in full_channel)\n",
    "    full_tokens = tokenize_with_spans(full_text)\n",
    "\n",
    "    all_results = []\n",
    "\n",
    "    for entry_index, entry in enumerate(side_channel):\n",
    "        channel_name = entry.get(\"channel\", \"unknown\")\n",
    "        base_id = f\"{channel_name}_{entry_index + 1}\"\n",
    "\n",
    "        # raw segments for this single original entry\n",
    "        raw_segments = align_one_entry(full_tokens, entry)\n",
    "\n",
    "        # If only one segment → treat as unsplit\n",
    "        if len(raw_segments) == 1:\n",
    "            seg = raw_segments[0]\n",
    "            seg_id = base_id\n",
    "            seg_parent_id = -1\n",
    "\n",
    "            seg_out = {\n",
    "                **seg,\n",
    "                \"channel\": channel_name,\n",
    "                \"seg_id\": seg_id,\n",
    "                \"seg_parent_id\": seg_parent_id,\n",
    "            }\n",
    "            all_results.append(seg_out)\n",
    "            continue\n",
    "\n",
    "        # Entry is split into multiple segments → assign child IDs\n",
    "        child_ids = [f\"{base_id}_{i + 1}\" for i in range(len(raw_segments))]\n",
    "\n",
    "        # Identify matched segments (anchors)\n",
    "        anchor_indices = [\n",
    "            i for i, seg in enumerate(raw_segments)\n",
    "            if seg[\"start_pos\"] >= 0 and seg[\"end_pos\"] >= 0\n",
    "        ]\n",
    "\n",
    "        for i, seg in enumerate(raw_segments):\n",
    "            seg_id = child_ids[i]\n",
    "            start_pos = seg[\"start_pos\"]\n",
    "            end_pos = seg[\"end_pos\"]\n",
    "\n",
    "            if start_pos >= 0 and end_pos >= 0:\n",
    "                # Matched segment → no parent\n",
    "                seg_parent_id = -1\n",
    "            else:\n",
    "                # Unmatched segment\n",
    "                parent_id = -1\n",
    "                if anchor_indices:\n",
    "                    if parent_preference == \"left\":\n",
    "                        # Prefer nearest matched segment to the LEFT, then RIGHT\n",
    "                        left_anchor = next(\n",
    "                            (idx for idx in reversed(anchor_indices) if idx < i),\n",
    "                            None\n",
    "                        )\n",
    "                        if left_anchor is not None:\n",
    "                            parent_id = child_ids[left_anchor]\n",
    "                        else:\n",
    "                            right_anchor = next(\n",
    "                                (idx for idx in anchor_indices if idx > i),\n",
    "                                None\n",
    "                            )\n",
    "                            if right_anchor is not None:\n",
    "                                parent_id = child_ids[right_anchor]\n",
    "                    else:\n",
    "                        # Default: prefer nearest matched segment to the RIGHT, then LEFT\n",
    "                        right_anchor = next(\n",
    "                            (idx for idx in anchor_indices if idx > i),\n",
    "                            None\n",
    "                        )\n",
    "                        if right_anchor is not None:\n",
    "                            parent_id = child_ids[right_anchor]\n",
    "                        else:\n",
    "                            left_anchor = next(\n",
    "                                (idx for idx in reversed(anchor_indices) if idx < i),\n",
    "                                None\n",
    "                            )\n",
    "                            if left_anchor is not None:\n",
    "                                parent_id = child_ids[left_anchor]\n",
    "                # If no anchors at all in this entry, parent_id stays -1\n",
    "                seg_parent_id = parent_id\n",
    "\n",
    "            seg_out = {\n",
    "                **seg,\n",
    "                \"channel\": channel_name,\n",
    "                \"seg_id\": seg_id,\n",
    "                \"seg_parent_id\": seg_parent_id,\n",
    "            }\n",
    "            all_results.append(seg_out)\n",
    "\n",
    "    return all_results\n",

    "# ---------- merge helper for seg_id parsing ----------\n",
    "\n",
    "def _parse_seg_id(seg_id: str):\n",
    "    \"\"\"\n",
    "    Parse seg_id into components:\n",
    "        \"left_3\"      -> (\"left\", 3, 0)\n",
    "        \"left_3_2\"    -> (\"left\", 3, 2)\n",
    "        \"right_10_5\"  -> (\"right\", 10, 5)\n",
    "    local_idx is 0 for unsplit entries.\n",
    "    \"\"\"\n",
    "    parts = seg_id.split(\"_\")\n",
    "    if len(parts) == 2:\n",
    "        channel = parts[0]\n",
    "        base_idx = int(parts[1])\n",
    "        local_idx = 0\n",
    "    elif len(parts) == 3:\n",
    "        channel = parts[0]\n",
    "        base_idx = int(parts[1])\n",
    "        local_idx = int(parts[2])\n",
    "    else:\n",
    "        # Fallback: treat everything after first as base, local 0\n",
    "        channel = parts[0]\n",
    "        base_idx = int(parts[1]) if len(parts) > 1 and parts[1].isdigit() else 0\n",
    "        local_idx = 0\n",
    "    return channel, base_idx, local_idx\n",

    "# ---------- merging function ----------\n",
    "\n",
    "def merge_aligned_segments(\n",
    "    left_segments,\n",
    "    right_segments,\n",
    "    left_channel_speaker: str = \"agent\",\n",
    "    right_channel_speaker: str = \"customer\",\n",
    "    merge_consecutive: bool = True,\n",
    "):\n",
    "    \"\"\"\n",
    "    Merge two lists of aligned segments (one from left channel, one from right channel)\n",
    "    into a single, time/position-ordered list.\n",
    "\n",
    "    - Segments with seg_parent_id == -1 and valid start_pos/end_pos act as \"anchors\".\n",
    "      They are sorted globally by start_pos (i.e., position in full_text).\n",
    "    - Dangling unmatched segments (start_pos == end_pos == -1 and seg_parent_id != -1)\n",
    "      appear adjacent to their parent anchor, with relative order preserved via seg_id.\n",
    "    - Independent unmatched segments (seg_parent_id == -1 and start_pos == -1)\n",
    "      are kept, placed after anchored blocks in original order.\n",
    "    - Output segments also contain:\n",
    "        \"channel\": \"left\" / \"right\"\n",
    "        \"speaker\": <left_channel_speaker> / <right_channel_speaker>\n",
    "    - If merge_consecutive is True (default), consecutive segments from the same\n",
    "      channel, same speaker, and same base seg_id are merged into one.\n",
    "    \"\"\"\n",
    "\n",
    "    # Ensure channel and speaker fields are set correctly from which list they came\n",
    "    for seg in left_segments:\n",
    "        seg[\"channel\"] = \"left\"\n",
    "        seg[\"speaker\"] = left_channel_speaker\n",
    "\n",
    "    for seg in right_segments:\n",
    "        seg[\"channel\"] = \"right\"\n",
    "        seg[\"speaker\"] = right_channel_speaker\n",
    "\n",
    "    # Combined list (we won't modify seg_id, seg_parent_id, positions, etc.)\n",
    "    combined = list(left_segments) + list(right_segments)\n",
    "\n",
    "    # Assign a stable merge order index per seg_id to preserve original ordering as tiebreaker\n",
    "    order_map = {}\n",
    "    seq = 0\n",
    "    for seg in combined:\n",
    "        seq += 1\n",
    "        order_map[seg[\"seg_id\"]] = seq\n",
    "\n",
    "    # Build mapping from parent_id -> list of children (dangling unmatched segments)\n",
    "    children_by_parent = {}\n",
    "    for seg in combined:\n",
    "        pid = seg.get(\"seg_parent_id\", -1)\n",
    "        if pid != -1:\n",
    "            children_by_parent.setdefault(pid, []).append(seg)\n",
    "\n",
    "    # Identify anchors: matched segments with no parent\n",
    "    anchors = []\n",
    "    for seg in combined:\n",
    "        if (\n",
    "            seg.get(\"seg_parent_id\", -1) == -1\n",
    "            and seg.get(\"start_pos\", -1) is not None\n",
    "            and seg[\"start_pos\"] >= 0\n",
    "            and seg.get(\"end_pos\", -1) is not None\n",
    "            and seg[\"end_pos\"] >= 0\n",
    "        ):\n",
    "            anchors.append(seg)\n",
    "\n",
    "    # Build blocks: each block = anchor + its children, sorted by local_idx\n",
    "    blocks = []\n",
    "    used_ids = set()\n",
    "\n",
    "    for anchor in anchors:\n",
    "        seg_id = anchor[\"seg_id\"]\n",
    "        kids = children_by_parent.get(seg_id, [])\n",
    "\n",
    "        group = [anchor] + kids\n",
    "\n",
    "        # Sort group within an original entry based on seg_id local_idx;\n",
    "        # this reconstructs the original split order so that dangling parts\n",
    "        # appear before/after the anchor correctly.\n",
    "        group.sort(key=lambda s: _parse_seg_id(s[\"seg_id\"])[2])\n",
    "\n",
    "        # Block ordering keys:\n",
    "        anchor_start_pos = anchor[\"start_pos\"]\n",
    "        block_order = min(order_map[s[\"seg_id\"]] for s in group)\n",
    "\n",
    "        blocks.append({\n",
    "            \"anchor_start_pos\": anchor_start_pos,\n",
    "            \"order\": block_order,\n",
    "            \"segments\": group,\n",
    "        })\n",
    "\n",
    "        for s in group:\n",
    "            used_ids.add(s[\"seg_id\"])\n",
    "\n",
    "    # Now handle independent segments:\n",
    "    # Those not in any block yet (not anchors, not children),\n",
    "    # or anchors that didn't match (start_pos == -1).\n",
    "    for seg in combined:\n",
    "        sid = seg[\"seg_id\"]\n",
    "        if sid in used_ids:\n",
    "            continue\n",
    "\n",
    "        start_pos = seg.get(\"start_pos\", -1)\n",
    "        if start_pos is None or start_pos < 0:\n",
    "            anchor_start_pos = float(\"inf\")  # push unmatched independents after all anchored blocks\n",
    "        else:\n",
    "            anchor_start_pos = start_pos\n",
    "\n",
    "        block = {\n",
    "            \"anchor_start_pos\": anchor_start_pos,\n",
    "            \"order\": order_map[sid],\n",
    "            \"segments\": [seg],\n",
    "        }\n",
    "        blocks.append(block)\n",
    "        used_ids.add(sid)\n",
    "\n",
    "    # Sort blocks:\n",
    "    #   1) by anchor_start_pos (anchored blocks first, then independents)\n",
    "    #   2) by original merge order as tiebreaker\n",
    "    blocks.sort(key=lambda b: (b[\"anchor_start_pos\"], b[\"order\"]))\n",
    "\n",
    "    # Flatten back to single list of segments\n",
    "    merged = []\n",
    "    for b in blocks:\n",
    "        merged.extend(b[\"segments\"])\n",
    "\n",
    "    # Optionally merge consecutive segments\n",
    "    if not merge_consecutive or not merged:\n",
    "        return merged\n",
    "\n",
    "    merged_consecutive = []\n",
    "\n",
    "    current = merged[0].copy()\n",
    "    cur_channel, cur_base, _ = _parse_seg_id(current[\"seg_id\"])\n",
    "    cur_speaker = current.get(\"speaker\")\n",
    "\n",
    "    current_start_time = current.get(\"start_time\", -1)\n",
    "    current_end_time = current.get(\"end_time\", -1)\n",
    "    current_start_pos = current.get(\"start_pos\", -1)\n",
    "    current_end_pos = current.get(\"end_pos\", -1)\n",
    "\n",
    "    for seg in merged[1:]:\n",
    "        ch, base, _ = _parse_seg_id(seg[\"seg_id\"])\n",
    "        speaker = seg.get(\"speaker\")\n",
    "\n",
    "        same_group = (\n",
    "            ch == cur_channel and\n",
    "            base == cur_base and\n",
    "            speaker == cur_speaker\n",
    "        )\n",
    "\n",
    "        if same_group:\n",
    "            # Merge into current\n",
    "            current[\"text\"] = current[\"text\"] + \" \" + seg[\"text\"]\n",
    "\n",
    "            st = seg.get(\"start_time\", -1)\n",
    "            et = seg.get(\"end_time\", -1)\n",
    "            if current_start_time == -1 and st != -1:\n",
    "                current_start_time = st\n",
    "            if et != -1:\n",
    "                current_end_time = et\n",
    "\n",
    "            sp = seg.get(\"start_pos\", -1)\n",
    "            ep = seg.get(\"end_pos\", -1)\n",
    "            if current_start_pos == -1 and sp != -1:\n",
    "                current_start_pos = sp\n",
    "            if ep != -1:\n",
    "                current_end_pos = ep\n",
    "\n",
    "            current[\"start_time\"] = current_start_time\n",
    "            current[\"end_time\"] = current_end_time\n",
    "            current[\"start_pos\"] = current_start_pos\n",
    "            current[\"end_pos\"] = current_end_pos\n",
    "            # seg_id and seg_parent_id stay from the first in group\n",
    "        else:\n",
